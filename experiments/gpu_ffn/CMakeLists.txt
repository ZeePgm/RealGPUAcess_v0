cmake_minimum_required(VERSION 3.24)
set(CMAKE_TRY_COMPILE_CONFIGURATION Release)
project(gpu_ffn LANGUAGES C CXX CUDA)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/adapters/llama_cpp/integrated" model)
find_package(CUDAToolkit REQUIRED)
find_library(FFN_CUPTI cupti HINTS "${CUDAToolkit_ROOT}/lib" "${CUDAToolkit_ROOT}/lib/x64" REQUIRED)
add_executable(gpu_ffn "${CMAKE_CURRENT_LIST_DIR}/native_s8.cpp")
target_compile_definitions(gpu_ffn PRIVATE GPU_S_JOINT=1 GPU_S_TRANSFER_TRACE=1 GPU_S_SEGMENTS=1 GPU_FFN_EXPERIMENT=1)
target_include_directories(gpu_ffn PRIVATE "${ROOT}/include" "${ROOT}/third_party/llama.cpp-integrated/vendor" "${CUDAToolkit_ROOT}/include")
target_link_libraries(gpu_ffn PRIVATE llama ggml CUDA::cudart "${FFN_CUPTI}" HBFEventLab::tensor_bridge)
if(MSVC)
  target_compile_options(gpu_ffn PRIVATE /utf-8 /EHsc)
endif()
add_executable(ffn_audit_tests audit_tests.cpp)
target_link_libraries(ffn_audit_tests PRIVATE HBFEventLab::core)
enable_testing()
add_test(NAME ffn_online_audit COMMAND ffn_audit_tests)
