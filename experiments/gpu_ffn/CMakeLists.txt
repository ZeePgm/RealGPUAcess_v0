cmake_minimum_required(VERSION 3.24)
project(gpu_ffn LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BASE_BUILD "${ROOT}/build-model" CACHE PATH "Verified dependency build")
set(LLAMA ${ROOT}/third_party/llama.cpp-integrated)
find_package(CUDAToolkit REQUIRED)
find_library(CUPTI cupti HINTS "${CUDAToolkit_ROOT}/lib" "${CUDAToolkit_ROOT}/lib/x64" REQUIRED)
add_executable(gpu_ffn ../gpu_s/native_s8.cpp)
target_compile_definitions(gpu_ffn PRIVATE GPU_S_JOINT=1 GPU_S_TRANSFER_TRACE=1 GPU_S_SEGMENTS=1 GPU_FFN_EXPERIMENT=1)
target_include_directories(gpu_ffn PRIVATE ${ROOT}/include ${LLAMA}/include ${LLAMA}/ggml/include ${LLAMA}/vendor ${ROOT}/third_party/HBF-EventLab/include)
target_link_libraries(gpu_ffn PRIVATE ${BASE_BUILD}/llama/src/llama.lib ${BASE_BUILD}/llama/ggml/src/ggml.lib ${BASE_BUILD}/llama/ggml/src/ggml-base.lib CUDA::cudart ${CUPTI} ${BASE_BUILD}/eventlab/hbf_eventlab_tensor_bridge.lib ${BASE_BUILD}/eventlab/hbf_eventlab.lib)
add_executable(ffn_audit_tests audit_tests.cpp)
target_include_directories(ffn_audit_tests PRIVATE ${ROOT}/third_party/HBF-EventLab/include ${LLAMA}/vendor)
target_link_libraries(ffn_audit_tests PRIVATE ${BASE_BUILD}/eventlab/hbf_eventlab.lib)
target_compile_options(gpu_ffn PRIVATE /utf-8 /EHsc)
target_compile_options(ffn_audit_tests PRIVATE /utf-8 /EHsc)
enable_testing()
add_test(NAME ffn_online_audit COMMAND ffn_audit_tests)
