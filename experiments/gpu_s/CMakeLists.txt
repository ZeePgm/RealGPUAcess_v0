cmake_minimum_required(VERSION 3.24)
project(gpu_s_stages LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BASE_BUILD "${ROOT}/build-portability-model" CACHE PATH "Verified workspace dependency build")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(LLAMA "${ROOT}/third_party/llama.cpp-integrated")
find_package(CUDAToolkit REQUIRED)
add_executable(gpu_s_native native.cpp)
target_include_directories(gpu_s_native PRIVATE "${ROOT}/include" "${LLAMA}/include" "${LLAMA}/ggml/include" "${LLAMA}/vendor")
target_link_libraries(gpu_s_native PRIVATE "${BASE_BUILD}/llama/src/llama.lib" "${BASE_BUILD}/llama/ggml/src/ggml.lib" "${BASE_BUILD}/llama/ggml/src/ggml-base.lib" CUDA::cudart)
find_library(CUPTI_LIBRARY cupti HINTS "${CUDAToolkit_ROOT}/lib" "${CUDAToolkit_ROOT}/lib/x64" REQUIRED)
target_link_libraries(gpu_s_native PRIVATE "${CUPTI_LIBRARY}")
target_include_directories(gpu_s_native PRIVATE "${CUDAToolkit_ROOT}/include")
add_executable(gpu_s_bridge bridge.cpp)
target_include_directories(gpu_s_bridge PRIVATE "${ROOT}/third_party/HBF-EventLab/include" "${LLAMA}/vendor")
target_link_libraries(gpu_s_bridge PRIVATE "${BASE_BUILD}/eventlab/hbf_eventlab_tensor_bridge.lib" "${BASE_BUILD}/eventlab/hbf_eventlab.lib")
add_executable(gpu_s_joint native.cpp)
target_compile_definitions(gpu_s_joint PRIVATE GPU_S_JOINT=1)
target_include_directories(gpu_s_joint PRIVATE "${ROOT}/include" "${LLAMA}/include" "${LLAMA}/ggml/include" "${LLAMA}/vendor" "${ROOT}/third_party/HBF-EventLab/include" "${CUDAToolkit_ROOT}/include")
target_link_libraries(gpu_s_joint PRIVATE "${BASE_BUILD}/llama/src/llama.lib" "${BASE_BUILD}/llama/ggml/src/ggml.lib" "${BASE_BUILD}/llama/ggml/src/ggml-base.lib" CUDA::cudart "${CUPTI_LIBRARY}" "${BASE_BUILD}/eventlab/hbf_eventlab_tensor_bridge.lib" "${BASE_BUILD}/eventlab/hbf_eventlab.lib")
add_executable(gpu_s_transfer_probe native.cpp)
target_compile_definitions(gpu_s_transfer_probe PRIVATE GPU_S_JOINT=1 GPU_S_TRANSFER_TRACE=1)
target_include_directories(gpu_s_transfer_probe PRIVATE "${ROOT}/include" "${LLAMA}/include" "${LLAMA}/ggml/include" "${LLAMA}/vendor" "${ROOT}/third_party/HBF-EventLab/include" "${CUDAToolkit_ROOT}/include")
target_link_libraries(gpu_s_transfer_probe PRIVATE "${BASE_BUILD}/llama/src/llama.lib" "${BASE_BUILD}/llama/ggml/src/ggml.lib" "${BASE_BUILD}/llama/ggml/src/ggml-base.lib" CUDA::cudart "${CUPTI_LIBRARY}" "${BASE_BUILD}/eventlab/hbf_eventlab_tensor_bridge.lib" "${BASE_BUILD}/eventlab/hbf_eventlab.lib")
add_executable(gpu_s_segmented native.cpp)
target_compile_definitions(gpu_s_segmented PRIVATE GPU_S_JOINT=1 GPU_S_TRANSFER_TRACE=1 GPU_S_SEGMENTS=1)
target_include_directories(gpu_s_segmented PRIVATE "${ROOT}/include" "${LLAMA}/include" "${LLAMA}/ggml/include" "${LLAMA}/vendor" "${ROOT}/third_party/HBF-EventLab/include" "${CUDAToolkit_ROOT}/include")
target_link_libraries(gpu_s_segmented PRIVATE "${BASE_BUILD}/llama/src/llama.lib" "${BASE_BUILD}/llama/ggml/src/ggml.lib" "${BASE_BUILD}/llama/ggml/src/ggml-base.lib" CUDA::cudart "${CUPTI_LIBRARY}" "${BASE_BUILD}/eventlab/hbf_eventlab_tensor_bridge.lib" "${BASE_BUILD}/eventlab/hbf_eventlab.lib")
if(MSVC)
  target_compile_options(gpu_s_segmented PRIVATE /utf-8 /EHsc)
  target_compile_options(gpu_s_native PRIVATE /utf-8 /EHsc)
  target_compile_options(gpu_s_bridge PRIVATE /utf-8 /EHsc)
  target_compile_options(gpu_s_joint PRIVATE /utf-8 /EHsc)
  target_compile_options(gpu_s_transfer_probe PRIVATE /utf-8 /EHsc)
endif()

add_executable(gpu_s_reference native_reference.cpp)
target_include_directories(gpu_s_reference PRIVATE "${ROOT}/include" "${LLAMA}/include" "${LLAMA}/ggml/include" "${LLAMA}/vendor" "${CUDAToolkit_ROOT}/include")
target_link_libraries(gpu_s_reference PRIVATE "${BASE_BUILD}/llama/src/llama.lib" "${BASE_BUILD}/llama/ggml/src/ggml.lib" "${BASE_BUILD}/llama/ggml/src/ggml-base.lib" CUDA::cudart "${CUPTI_LIBRARY}")
if(MSVC)
 target_compile_options(gpu_s_reference PRIVATE /utf-8 /EHsc)
endif()

add_executable(gpu_s_s8 native_s8.cpp)
target_compile_definitions(gpu_s_s8 PRIVATE GPU_S_JOINT=1 GPU_S_TRANSFER_TRACE=1 GPU_S_SEGMENTS=1)
target_include_directories(gpu_s_s8 PRIVATE "${ROOT}/include" "${LLAMA}/include" "${LLAMA}/ggml/include" "${LLAMA}/vendor" "${ROOT}/third_party/HBF-EventLab/include" "${CUDAToolkit_ROOT}/include")
target_link_libraries(gpu_s_s8 PRIVATE "${BASE_BUILD}/llama/src/llama.lib" "${BASE_BUILD}/llama/ggml/src/ggml.lib" "${BASE_BUILD}/llama/ggml/src/ggml-base.lib" CUDA::cudart "${CUPTI_LIBRARY}" "${BASE_BUILD}/eventlab/hbf_eventlab_tensor_bridge.lib" "${BASE_BUILD}/eventlab/hbf_eventlab.lib")
if(MSVC)
 target_compile_options(gpu_s_s8 PRIVATE /utf-8 /EHsc)
endif()
